\section{Apresentação da Matéria}
\begin{itemize}[label=$\rightarrow$]
    \item 136 horas aula
    \item Disciplina focada na elaboração e execução de ações de formação e 
    capacidade tecnológica voltadas diretamente para a comunidade
\end{itemize}

    \subsection{Os três pilares da disciplina}
    \begin{enumerate}
        \item Identificar -- Mapear demandas reais e oportunidades de letramento 
        tecnológica na comunidade
        \item Planejar -- Estruturar ações de capacitação em informática sob medida 
        para o público alvo
        \item Executar -- Colocar o projeto em prática, entregando valor e formação 
        técnica
    \end{enumerate}

    \subsection{Mapa do Projeto}
    \begin{enumerate}
        \item Diagnóstico -- Descobrir a necessidade
        \item Planejamento -- Desenhar a solução
        \item Execução -- Entregar o resultado
    \end{enumerate}
