# Valor de posição do fibonacci (0...)
# M[0] = Posição que deseja da sequencia
# M[1] = Penultimo valor
# M[2] = Ultimo valor
# M[3] = Contador
# M[4] = Constante 1 (incremento e verificação)
# M[5] = Resultado

30 # Resultado esperado: 3
0
1
0
1

# Instruções
# Contagem (M[3] - M[0], enquanto for <= 0 roda o código)
LOAD M(3)
SUB M(0) 
JUMP +M(112) # Se for 0 acaba o código

# Verifica se está na repetição 1
LOAD M(3)
SUB M(4)
JUMP +M(104, 20:39)
LOAD M(4)
STOR M(5)
JUMP M(108, 0:19)

LOAD M(1) # M(104, 20:39)
ADD M(2)

# Coloca o resultado no M[5]
STOR M(5)

# Atualiza M[1] e M[2]
LOAD M(2)
STOR M(1)
LOAD M(5)
STOR M(2)

# Incrementa M[3]
LOAD M(3) # M[108]
ADD M(4)
STOR M(3)
JUMP M(100, 0:19)
